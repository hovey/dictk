6def _prepare( 

7 *, kernel: np.ndarray, search: np.ndarray 

8) -> tuple[np.ndarray, np.ndarray]: 

9 """Validate `kernel`/`search` and cast both to float64. 

10 

11 Args: 

12 kernel: The fixed template subimage. 

13 search: The larger subimage to slide `kernel` across. 

14 

15 Returns: 

16 `(kernel, search)`, both cast to float64. 

17 

18 Raises: 

19 ValueError: If either array is not 2D, or `search` is smaller than 

20 `kernel` in either dimension. 

21 """ 

22 if kernel.ndim != 2: 

23 raise ValueError(f"kernel must be 2D, got shape {kernel.shape}") 

24 if search.ndim != 2: 

25 raise ValueError(f"search must be 2D, got shape {search.shape}") 

26 if search.shape[0] < kernel.shape[0] or search.shape[1] < kernel.shape[1]: 

27 raise ValueError( 

28 f"search shape {search.shape} must be >= kernel shape {kernel.shape} " 

29 "in both dimensions" 

30 ) 

31 return kernel.astype(np.float64), search.astype(np.float64)

34def _safe_divide(*, numerator: np.ndarray, denominator: np.ndarray) -> np.ndarray: 

35 """Elementwise `numerator / denominator`, substituting 0 wherever `denominator` is 0. 

36 

37 Args: 

38 numerator: Values to divide. 

39 denominator: Values to divide by; must be broadcastable against 

40 `numerator` and non-negative (as with a square root of an 

41 energy term). 

42 

43 Returns: 

44 `numerator / denominator`, with 0 wherever `denominator` is 0 

45 (avoiding a division-by-zero warning, not just its result). 

46 """ 

47 safe_denominator = np.where(denominator == 0, 1.0, denominator) 

48 return np.where(denominator == 0, 0.0, numerator / safe_denominator)

51def _windows(*, search: np.ndarray, kernel_shape: tuple[int, int]) -> np.ndarray: 

52 """Return every `kernel_shape`-sized window of `search`, as one strided view. 

53 

54 Args: 

55 search: The 2D array to slide a window across. 

56 kernel_shape: The `(height, width)` of each window. 

57 

58 Returns: 

59 A 4D array of shape `(out_height, out_width, *kernel_shape)`, where 

60 `out_height = search.shape[0] - kernel_shape[0] + 1` and likewise for 

61 `out_width`. Entry `[dy, dx]` is the `kernel_shape`-sized window of 

62 `search` with its own top-left corner at offset `(dx, dy)`. 

63 """ 

64 return np.lib.stride_tricks.sliding_window_view(search, kernel_shape)

67def cc(*, kernel: np.ndarray, search: np.ndarray) -> np.ndarray: 

68 r"""Cross-correlation (CC) surface of `kernel` slid over `search`. 

69 

70 At every valid position, computes $C_{\rm CC} = \sum f_i g_i$, where 

71 $f$ is `kernel` and $g$ is the same-sized window of `search` at that 

72 position. Robust to neither brightness nor contrast differences between 

73 `kernel` and `search` — a uniform offset or scaling of either changes 

74 every value. 

75 

76 See Pan B, Xie H, Wang Z. "Equivalence of digital image correlation 

77 criteria for pattern matching." Applied Optics 2010;49(28):5501-9. 

78 

79 Args: 

80 kernel: The fixed template subimage (`f`). 

81 search: The larger subimage to slide `kernel` across (`g`'s source). 

82 

83 Returns: 

84 A 2D float64 array of shape 

85 `(search.shape[0] - kernel.shape[0] + 1, search.shape[1] - kernel.shape[1] + 1)`. 

86 Entry `[dy, dx]` is $C_{\rm CC}$ with `kernel`'s top-left corner at 

87 offset `(dx, dy)` in `search`'s local frame. 

88 

89 Raises: 

90 ValueError: If either array is not 2D, or `search` is smaller than 

91 `kernel` in either dimension. 

92 """ 

93 kernel, search = _prepare(kernel=kernel, search=search) 

94 windows = _windows(search=search, kernel_shape=kernel.shape) 

95 return (windows * kernel).sum(axis=(-2, -1))

98def ncc(*, kernel: np.ndarray, search: np.ndarray) -> np.ndarray: 

99 r"""Normalized cross-correlation (NCC) surface of `kernel` slid over `search`. 

100 

101 At every valid position, computes 

102 $C_{\rm NCC} = \sum f_i g_i \,/\, \sqrt{\sum f_i^2 \sum g_i^2}$, where 

103 $f$ is `kernel` and $g$ is the same-sized window of `search` at that 

104 position. Robust to a uniform contrast (multiplicative) difference 

105 between `kernel` and `search`, since scaling either side by a positive 

106 constant cancels between the numerator and denominator. Not robust to 

107 brightness (additive) differences. A window with zero energy (e.g. a 

108 flat, constant-valued region) contributes a value of 0 rather than 

109 raising a division-by-zero error. 

110 

111 See Pan B, Xie H, Wang Z. "Equivalence of digital image correlation 

112 criteria for pattern matching." Applied Optics 2010;49(28):5501-9. 

113 

114 Args: 

115 kernel: The fixed template subimage (`f`). 

116 search: The larger subimage to slide `kernel` across (`g`'s source). 

117 

118 Returns: 

119 A 2D float64 array of shape 

120 `(search.shape[0] - kernel.shape[0] + 1, search.shape[1] - kernel.shape[1] + 1)`. 

121 Entry `[dy, dx]` is $C_{\rm NCC}$ with `kernel`'s top-left corner at 

122 offset `(dx, dy)` in `search`'s local frame. 

123 

124 Raises: 

125 ValueError: If either array is not 2D, or `search` is smaller than 

126 `kernel` in either dimension. 

127 """ 

128 kernel, search = _prepare(kernel=kernel, search=search) 

129 windows = _windows(search=search, kernel_shape=kernel.shape) 

130 numerator = (windows * kernel).sum(axis=(-2, -1)) 

131 kernel_energy = np.sum(kernel**2) 

132 window_energy = (windows**2).sum(axis=(-2, -1)) 

133 denominator = np.sqrt(kernel_energy * window_energy) 

134 return _safe_divide(numerator=numerator, denominator=denominator)

137def zcc(*, kernel: np.ndarray, search: np.ndarray) -> np.ndarray: 

138 r"""Zero-mean cross-correlation (ZCC) surface of `kernel` slid over `search`. 

139 

140 At every valid position, computes 

141 $C_{\rm ZCC} = \sum (f_i - \bar{f})(g_i - \bar{g})$, where $f$ is 

142 `kernel`, $g$ is the same-sized window of `search` at that position, 

143 $\bar{f}$ is `kernel`'s own mean (fixed across all positions, since the 

144 kernel never moves), and $\bar{g}$ is that window's own local mean 

145 (recomputed at every position, not a global `search` statistic). Robust 

146 to a uniform brightness (additive) difference between `kernel` and 

147 `search`, since subtracting each side's own local mean cancels any 

148 constant added to that side. Not robust to contrast (multiplicative) 

149 differences. 

150 

151 See Pan B, Xie H, Wang Z. "Equivalence of digital image correlation 

152 criteria for pattern matching." Applied Optics 2010;49(28):5501-9. 

153 

154 Args: 

155 kernel: The fixed template subimage (`f`). 

156 search: The larger subimage to slide `kernel` across (`g`'s source). 

157 

158 Returns: 

159 A 2D float64 array of shape 

160 `(search.shape[0] - kernel.shape[0] + 1, search.shape[1] - kernel.shape[1] + 1)`. 

161 Entry `[dy, dx]` is $C_{\rm ZCC}$ with `kernel`'s top-left corner at 

162 offset `(dx, dy)` in `search`'s local frame. 

163 

164 Raises: 

165 ValueError: If either array is not 2D, or `search` is smaller than 

166 `kernel` in either dimension. 

167 """ 

168 kernel, search = _prepare(kernel=kernel, search=search) 

169 windows = _windows(search=search, kernel_shape=kernel.shape) 

170 kernel_centered = kernel - kernel.mean() 

171 windows_centered = windows - windows.mean(axis=(-2, -1), keepdims=True) 

172 return (windows_centered * kernel_centered).sum(axis=(-2, -1))

175def zncc(*, kernel: np.ndarray, search: np.ndarray) -> np.ndarray: 

176 r"""Zero-mean normalized cross-correlation (ZNCC) surface of `kernel` slid over `search`. 

177 

178 At every valid position, computes 

179 $C_{\rm ZNCC} = \sum \bar{f}_i \bar{g}_i \,/\, \sqrt{\sum \bar{f}_i^2 \sum \bar{g}_i^2}$, 

180 where $\bar{f}_i = f_i - \bar{f}$ and $\bar{g}_i = g_i - \bar{g}$ ($f$ = 

181 `kernel`, $g$ = the same-sized window of `search` at that position, 

182 $\bar{f}$/$\bar{g}$ their respective means -- $\bar{f}$ fixed, $\bar{g}$ 

183 recomputed locally per position, as in `zcc`). Robust to both brightness 

184 (additive) and contrast (multiplicative) differences between `kernel` 

185 and `search`, combining `zcc`'s brightness invariance with `ncc`'s 

186 contrast invariance. A window with zero variance (e.g. a flat, 

187 constant-valued region) contributes a value of 0 rather than raising a 

188 division-by-zero error. 

189 

190 See Pan B, Xie H, Wang Z. "Equivalence of digital image correlation 

191 criteria for pattern matching." Applied Optics 2010;49(28):5501-9. 

192 

193 Args: 

194 kernel: The fixed template subimage (`f`). 

195 search: The larger subimage to slide `kernel` across (`g`'s source). 

196 

197 Returns: 

198 A 2D float64 array of shape 

199 `(search.shape[0] - kernel.shape[0] + 1, search.shape[1] - kernel.shape[1] + 1)`. 

200 Entry `[dy, dx]` is $C_{\rm ZNCC}$ with `kernel`'s top-left corner at 

201 offset `(dx, dy)` in `search`'s local frame. 

202 

203 Raises: 

204 ValueError: If either array is not 2D, or `search` is smaller than 

205 `kernel` in either dimension. 

206 """ 

207 kernel, search = _prepare(kernel=kernel, search=search) 

208 windows = _windows(search=search, kernel_shape=kernel.shape) 

209 kernel_centered = kernel - kernel.mean() 

210 windows_centered = windows - windows.mean(axis=(-2, -1), keepdims=True) 

211 numerator = (windows_centered * kernel_centered).sum(axis=(-2, -1)) 

212 kernel_energy = np.sum(kernel_centered**2) 

213 window_energy = (windows_centered**2).sum(axis=(-2, -1)) 

214 denominator = np.sqrt(kernel_energy * window_energy) 

215 return _safe_divide(numerator=numerator, denominator=denominator)
