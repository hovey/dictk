  62def subimage(
  63    *, image: np.ndarray, origin: PixelCoordinate, width: int, height: int
  64) -> np.ndarray:
  65    """Extract a width x height crop of `image` with its top-left corner at `origin`.
  66
  67    `origin` is expressed in `image`'s own pixel reference frame (`image`'s
  68    top-left corner is `(0, 0)`) and may lie partially or entirely outside
  69    `image`'s bounds. Wherever the requested region doesn't overlap
  70    `image`, the corresponding output pixels are zero (black) rather than
  71    raising an error — so a subimage straddling an edge, or lying
  72    completely outside `image`, is always a valid, well-shaped result.
  73
  74    Args:
  75        image: Source 2D grayscale image array.
  76        origin: Top-left corner of the region to extract, in `image`'s
  77            pixel reference frame.
  78        width: Width of the extracted region in pixels. Must be >= 1.
  79        height: Height of the extracted region in pixels. Must be >= 1.
  80
  81    Returns:
  82        A 2D array of shape (height, width), same dtype as `image`.
  83
  84    Raises:
  85        ValueError: If `image` is not 2D, or width or height is less than 1.
  86    """
  87    if image.ndim != 2:
  88        raise ValueError(f"image must be 2D, got shape {image.shape}")
  89    if width < 1:
  90        raise ValueError(f"width {width} must be >= 1")
  91    if height < 1:
  92        raise ValueError(f"height {height} must be >= 1")
  93
  94    image_height, image_width = image.shape
  95    result = np.zeros((height, width), dtype=image.dtype)
  96
  97    src_x_start = max(origin.x, 0)
  98    src_y_start = max(origin.y, 0)
  99    src_x_end = min(origin.x + width, image_width)
 100    src_y_end = min(origin.y + height, image_height)
 101
 102    if src_x_start >= src_x_end or src_y_start >= src_y_end:
 103        return result  # requested region doesn't overlap image at all
 104
 105    dst_x_start = src_x_start - origin.x
 106    dst_y_start = src_y_start - origin.y
 107    dst_x_end = dst_x_start + (src_x_end - src_x_start)
 108    dst_y_end = dst_y_start + (src_y_end - src_y_start)
 109
 110    result[dst_y_start:dst_y_end, dst_x_start:dst_x_end] = image[
 111        src_y_start:src_y_end, src_x_start:src_x_end
 112    ]
 113    return result

 116def subimage_bounds_plot(
 117    *,
 118    image: np.ndarray,
 119    origin: PixelCoordinate,
 120    width: int,
 121    height: int,
 122    path: Path,
 123    dpi: int = 300,
 124) -> None:
 125    """Save a figure overlaying a source image's bounds and a subimage region.
 126
 127    Draws `image`'s own bounds in blue and the requested `origin`/`width`/
 128    `height` region in red, on top of `image` itself — including cases
 129    where the red region extends partially or completely outside the blue
 130    one, to visualize what `subimage()` would crop from. An `'o'` marker
 131    is also drawn at each rectangle's own origin: blue at `(0, 0)` for
 132    `image`, red at `origin` for the subimage — both in `image`'s pixel
 133    reference frame.
 134
 135    Args:
 136        image: Source 2D grayscale image array.
 137        origin: Top-left corner of the region, in `image`'s pixel
 138            reference frame; see `subimage()`.
 139        width: Width of the region in pixels. Must be >= 1.
 140        height: Height of the region in pixels. Must be >= 1.
 141        path: Output file path for the figure; format is inferred from
 142            the extension by matplotlib's savefig (e.g. .png), not
 143            dictk's own write/write_svg.
 144        dpi: Resolution of the saved figure.
 145
 146    Raises:
 147        ValueError: If width or height is less than 1.
 148    """
 149    if width < 1:
 150        raise ValueError(f"width {width} must be >= 1")
 151    if height < 1:
 152        raise ValueError(f"height {height} must be >= 1")
 153
 154    image_height, image_width = image.shape
 155
 156    margin = max(width, height, image_width, image_height) * 0.05
 157    x_min = min(0, origin.x) - margin
 158    x_max = max(image_width, origin.x + width) + margin
 159    y_min = min(0, origin.y) - margin
 160    y_max = max(image_height, origin.y + height) + margin
 161
 162    figsize = (
 163        (x_max - x_min) / _FIGURE_PIXELS_PER_INCH,
 164        (y_max - y_min) / _FIGURE_PIXELS_PER_INCH,
 165    )
 166    fig, ax = plt.subplots(figsize=figsize)
 167    ax.imshow(
 168        image, cmap="gray", origin="upper", extent=(0, image_width, image_height, 0)
 169    )
 170
 171    ax.add_patch(
 172        patches.Rectangle(
 173            (0, 0),
 174            image_width,
 175            image_height,
 176            edgecolor="blue",
 177            facecolor="none",
 178            linewidth=1.5,
 179            label="source image",
 180        )
 181    )
 182    ax.add_patch(
 183        patches.Rectangle(
 184            (origin.x, origin.y),
 185            width,
 186            height,
 187            edgecolor="red",
 188            facecolor="none",
 189            linewidth=1.5,
 190            label="subimage",
 191        )
 192    )
 193    ax.plot(0, 0, marker="o", color="blue", markersize=6)
 194    ax.plot(origin.x, origin.y, marker="o", color="red", markersize=6)
 195
 196    ax.set_xlim(x_min, x_max)
 197    ax.set_ylim(y_max, y_min)  # inverted: image y increases downward
 198
 199    ax.set_xlabel("x (pixels)")
 200    ax.set_ylabel("y (pixels)")
 201    # Legend placed fully outside the axes (not just an "upper right"-style
 202    # corner) so it never overlaps the source or subimage rectangles,
 203    # regardless of how much of the frame they fill; bbox_inches="tight"
 204    # on save keeps it from being clipped off the saved figure.
 205    ax.legend(loc="center left", bbox_to_anchor=(1.02, 0.5), fontsize=8)
 206    ax.set_title(f"subimage bounds: origin=({origin.x}, {origin.y}), {width}x{height}")
 207
 208    plt.savefig(path, dpi=dpi, bbox_inches="tight")
 209    plt.close(fig)

 285def subimage_comparison_plot(
 286    *,
 287    image: np.ndarray,
 288    origin: PixelCoordinate,
 289    width: int,
 290    height: int,
 291    path: Path,
 292    point: PixelCoordinate | None = None,
 293    point_color: str = "gold",
 294    point_label: str | None = None,
 295    subimage_label: str | None = None,
 296    color: str = "red",
 297    origin_label: str | None = None,
 298    source_origin_label: str | None = None,
 299    figsize: tuple[float, float] | None = None,
 300    dpi: int = 300,
 301) -> None:
 302    """Save a side-by-side comparison of a subimage's placement and its extraction.
 303
 304    The left panel matches `subimage_bounds_plot()`: `image`'s bounds in
 305    blue, the requested region in `color` (red by default), with an `'o'`
 306    marker at each rectangle's own origin. The right panel shows the
 307    extracted subimage (via `subimage()`) with a `color` border and
 308    origin marker, in its own local reference frame — but drawn using the
 309    *same* axis limits as the left panel, rather than being cropped or
 310    zoomed to the subimage's own size the way `subimage_plot()` is. That
 311    shared scale is what makes the two boxes render at identical size by
 312    construction — same data units per pixel in both panels — rather than
 313    approximating it by sizing each panel's figure independently around
 314    its own content.
 315
 316    Args:
 317        image: Source 2D grayscale image array.
 318        origin: Top-left corner of the region, in `image`'s pixel
 319            reference frame; see `subimage()`.
 320        width: Width of the region in pixels. Must be >= 1.
 321        height: Height of the region in pixels. Must be >= 1.
 322        path: Output file path for the figure; format is inferred from
 323            the extension by matplotlib's savefig (e.g. .png), not
 324            dictk's own write/write_svg.
 325        point: An optional point of interest, in `image`'s pixel reference
 326            frame, marked with a `point_color` dot on both panels — at
 327            `point` itself on the left, and at `point` translated into
 328            the subimage's own local frame (`point` minus `origin`) on
 329            the right, since that's the same physical point expressed in
 330            each panel's own coordinates.
 331        point_color: Matplotlib color name for `point`'s marker. Ignored
 332            if `point` isn't given.
 333        point_label: An optional short text label (e.g. `"P"`) drawn next
 334            to `point`'s marker, on both panels. Ignored if `point` isn't
 335            given.
 336        subimage_label: An optional short label identifying what this
 337            subimage represents (e.g. `"kernel"`), appended to the right
 338            panel's "subimage" title as "subimage (label)". Left as-is
 339            ("subimage") when not given.
 340        color: Matplotlib color name for the subimage's bounding box and
 341            origin marker, on both panels.
 342        origin_label: An optional short text label (e.g. `"K"`) drawn
 343            next to the subimage's own origin marker, on both panels.
 344        source_origin_label: An optional short text label (e.g. `"O"`)
 345            drawn next to `image`'s own origin marker (the blue dot),
 346            on the left panel only -- the right panel has no equivalent
 347            marker for it.
 348        figsize: Optional (width, height) in inches for the saved figure
 349            (both panels combined). By default the canvas is sized from
 350            `image`/the subimage's own data extent (so a small subimage
 351            yields a small figure); pass this to override with a fixed
 352            size instead -- e.g. matplotlib's own default, `(6.4, 4.8)`,
 353            where its default font sizes and line widths look as intended.
 354        dpi: Resolution of the saved figure.
 355
 356    Raises:
 357        ValueError: If width or height is less than 1.
 358    """
 359    if width < 1:
 360        raise ValueError(f"width {width} must be >= 1")
 361    if height < 1:
 362        raise ValueError(f"height {height} must be >= 1")
 363
 364    region = subimage(image=image, origin=origin, width=width, height=height)
 365    image_height, image_width = image.shape
 366
 367    margin = max(width, height, image_width, image_height) * 0.05
 368    # Small up-right offset for marker labels, so label text doesn't sit
 369    # directly on top of its own marker.
 370    label_offset = max(image_width, image_height) * 0.03
 371    label_font_size = 12
 372    # source_origin_label/origin_label sit just up-right of a marker at
 373    # (0, 0) -- the top-left corner of the panel's own content -- so only
 374    # the top margin (where a label there would land, above the panel
 375    # title) needs widening to fit one; left/right/bottom stay at the
 376    # default margin instead of also carrying that extra, unused space.
 377    # Anchored so the label's own bottom edge sits `label_offset` above
 378    # the marker (matching the gap already below the marker, down to the
 379    # box edge at y=0) and its top edge sits that same `label_offset`
 380    # below the panel's own top border -- the same gap on both sides of
 381    # the label, rather than a fixed multiple of the font size that
 382    # doesn't relate to the marker's own gap below.
 383    top_margin = margin
 384    if source_origin_label is not None or origin_label is not None:
 385        label_text_height = label_font_size / 72 * _FIGURE_PIXELS_PER_INCH
 386        top_margin = max(margin, 2 * label_offset + label_text_height)
 387    x_min = min(0, origin.x) - margin
 388    x_max = max(image_width, origin.x + width) + margin
 389    y_min = min(0, origin.y) - top_margin
 390    y_max = max(image_height, origin.y + height) + margin
 391
 392    # A thin white outline keeps every marker label legible against
 393    # whatever's directly behind it in the image content.
 394    label_outline = [patheffects.withStroke(linewidth=1, foreground="white")]
 395
 396    if figsize is None:
 397        panel_width = (x_max - x_min) / _FIGURE_PIXELS_PER_INCH
 398        panel_height = (y_max - y_min) / _FIGURE_PIXELS_PER_INCH
 399        figsize = (panel_width * 2, panel_height)
 400    fig, (ax_left, ax_right) = plt.subplots(
 401        1, 2, figsize=figsize, constrained_layout=True
 402    )
 403
 404    ax_left.imshow(
 405        image, cmap="gray", origin="upper", extent=(0, image_width, image_height, 0)
 406    )
 407    ax_left.add_patch(
 408        patches.Rectangle(
 409            (0, 0),
 410            image_width,
 411            image_height,
 412            edgecolor="blue",
 413            facecolor="none",
 414            linewidth=1.5,
 415        )
 416    )
 417    ax_left.add_patch(
 418        patches.Rectangle(
 419            (origin.x, origin.y),
 420            width,
 421            height,
 422            edgecolor=color,
 423            facecolor="none",
 424            linewidth=1.5,
 425        )
 426    )
 427    ax_left.plot(0, 0, marker="o", color="blue", markersize=6)
 428    ax_left.plot(origin.x, origin.y, marker="o", color=color, markersize=6)
 429    if point is not None:
 430        ax_left.plot(point.x, point.y, marker="o", color=point_color, markersize=6)
 431    if source_origin_label is not None:
 432        ax_left.text(
 433            label_offset,
 434            -label_offset,
 435            source_origin_label,
 436            color="blue",
 437            fontsize=label_font_size,
 438            va="bottom",
 439            path_effects=label_outline,
 440        )
 441    if origin_label is not None:
 442        ax_left.text(
 443            origin.x + label_offset,
 444            origin.y - label_offset,
 445            origin_label,
 446            color=color,
 447            fontsize=label_font_size,
 448            va="bottom",
 449            path_effects=label_outline,
 450        )
 451    if point is not None and point_label is not None:
 452        ax_left.text(
 453            point.x + label_offset,
 454            point.y - label_offset,
 455            point_label,
 456            color=point_color,
 457            fontsize=12,
 458            path_effects=label_outline,
 459        )
 460    ax_left.set_xlim(x_min, x_max)
 461    ax_left.set_ylim(y_max, y_min)  # inverted: image y increases downward
 462    ax_left.set_xlabel("x (pixels)")
 463    ax_left.set_ylabel("y (pixels)")
 464    ax_left.set_title("source image + subimage")
 465
 466    ax_right.imshow(region, cmap="gray", origin="upper", extent=(0, width, height, 0))
 467    ax_right.add_patch(
 468        patches.Rectangle(
 469            (0, 0),
 470            width,
 471            height,
 472            edgecolor=color,
 473            facecolor="none",
 474            linewidth=1.5,
 475        )
 476    )
 477    ax_right.plot(0, 0, marker="o", color=color, markersize=6)
 478    if point is not None:
 479        ax_right.plot(
 480            point.x - origin.x,
 481            point.y - origin.y,
 482            marker="o",
 483            color=point_color,
 484            markersize=6,
 485        )
 486    if origin_label is not None:
 487        ax_right.text(
 488            label_offset,
 489            -label_offset,
 490            origin_label,
 491            color=color,
 492            fontsize=label_font_size,
 493            va="bottom",
 494            path_effects=label_outline,
 495        )
 496    if point is not None and point_label is not None:
 497        ax_right.text(
 498            point.x - origin.x + label_offset,
 499            point.y - origin.y - label_offset,
 500            point_label,
 501            color=point_color,
 502            fontsize=12,
 503            path_effects=label_outline,
 504        )
 505    # Same limits as ax_left, not the subimage's own tight size -- this is
 506    # the whole point: identical data-units-per-pixel in both panels.
 507    ax_right.set_xlim(x_min, x_max)
 508    ax_right.set_ylim(y_max, y_min)
 509    ax_right.set_xlabel("x (pixels)")
 510    ax_right.set_ylabel("y (pixels)")
 511    ax_right.set_title(
 512        f"subimage ({subimage_label})" if subimage_label is not None else "subimage"
 513    )
 514
 515    plt.savefig(path, dpi=dpi, bbox_inches="tight")
 516    plt.close(fig)

 643    # default margin instead of also carrying that extra, unused space.

 645    # its marker (matching the gap already below the marker, e.g. down
 646    # to a box edge at y=0) and its top edge sits that same

 648    # on both sides of the label, rather than a fixed multiple of the
 649    # font size that doesn't relate to the marker's own gap below.
 650    top_margin = margin

 652        label_text_height = label_font_size / 72 * _FIGURE_PIXELS_PER_INCH

 660    if figsize is None:
 661        figsize = (
 662            (x_max - x_min) / _FIGURE_PIXELS_PER_INCH,
 663            (y_max - y_min) / _FIGURE_PIXELS_PER_INCH,

 672            patches.Rectangle(
 673                (box.origin.x, box.origin.y),

 704            fontsize=label_font_size,
 705            va="bottom",

 733def reference_frame_plot(*, image: np.ndarray, path: Path, dpi: int = 300) -> None:
 734    """Save a side-by-side figure contrasting `image` with its reference frame made explicit.
 735
 736    Left panel: `image` alone, with no annotation — no axes box, ticks, or
 737    labels either — a reminder that a pixel reference frame is always

 739    panel: the same `image`, with its bounds outlined in blue and a blue
 740    'o' marker at its origin `(0, 0)` (top-left corner) — the same style
 741    `subimage_bounds_plot()` uses for a source image — plus a red arrow
 742    along the x-axis and a green arrow along the y-axis, both starting
 743    from that origin, making the top-left-origin, y-increases-downward
 744    pixel convention used throughout this codebase explicit, labeled with
 745    the calligraphic frame symbol "F" just above and to the left of the
 746    origin marker. The "x"/"y" axis labels sit in the margin outside the
 747    blue frame, rather than on top of it, so they don't overlap its
 748    border.

 750    Args:
 751        image: Source 2D grayscale image array.
 752        path: Output file path for the figure; format is inferred from
 753            the extension by matplotlib's savefig (e.g. .png), not
 754            dictk's own write/write_svg.

 756    """

 759
 760    margin = max(image_width, image_height) * 0.05
 761    # The "x"/"y" labels need enough margin to sit clear of both the blue
 762    # frame and the axes spine; the label's rendered size only depends on
 763    # its font size and the data-units-per-inch scale figsize is built
 764    # from, not on image size, so a small image's default margin (5% of
 765    # its own dimensions) can otherwise be too little to fit it.

 767    frame_label_font_size = 14

 769        max(label_font_size, frame_label_font_size) * 2.2 / 72 * _FIGURE_PIXELS_PER_INCH
 770    )
 771    margin = max(margin, min_margin_for_labels)
 772
 773    x_min, x_max = -margin, image_width + margin
 774    y_min, y_max = -margin, image_height + margin
 775
 776    panel_width = (x_max - x_min) / _FIGURE_PIXELS_PER_INCH
 777    panel_height = (y_max - y_min) / _FIGURE_PIXELS_PER_INCH

 783        ax.imshow(
 784            image,
 785            cmap="gray",
 786            origin="upper",
 787            extent=(0, image_width, image_height, 0),
 788        )
 789        ax.set_xlim(x_min, x_max)
 790        ax.set_ylim(y_max, y_min)  # inverted: image y increases downward

 792    ax_left.set_title("image")

 794
 795    ax_right.set_xlabel("x (pixels)")
 796    ax_right.set_ylabel("y (pixels)")
 797
 798    ax_right.add_patch(
 799        patches.Rectangle(

 803            edgecolor="blue",
 804            facecolor="none",
 805            linewidth=1.5,
 806        )
 807    )

 812    frame_label_offset = margin * 0.35

 814        -frame_label_offset,
 815        -frame_label_offset,
 816        r"$\mathcal{F}$",
 817        color="blue",
 818        fontsize=frame_label_font_size,
 819        ha="right",
 820        va="bottom",

 830            (axis_length, 0),
 831            (axis_length, -label_offset),
 832            "red",
 833            "x",
 834            "center",
 835            "bottom",
 836        ),
 837        (
 838            (0, axis_length),
 839            (-label_offset, axis_length),
 840            "green",
 841            "y",
 842            "right",

 847            "",

 849            xytext=(0, 0),

 852        ax_right.text(
 853            *label_pos, label, color=color, fontsize=label_font_size, ha=ha, va=va

 862def checkerboard(
 863    *, width: int, height: int, count_x: int = 8, count_y: int = 8
 864) -> np.ndarray:
 865    """Generate a black-and-white checkerboard test image.
 866
 867    Same parameters and pixel values as the `dictk checkerboard` CLI
 868    command, minus the file write — see the `dictk` package docstring for
 869    why the API layer stops at the array. count_x and count_y are named for
 870    the rectangles they divide the image into, not "squares": unequal
 871    counts (or a width/height ratio that doesn't match count_x/count_y)
 872    produce rectangular cells, not square ones.
 873
 874    Args:
 875        width: Image width in pixels.
 876        height: Image height in pixels.
 877        count_x: Number of rectangles along the width. Must be >= 1.
 878        count_y: Number of rectangles along the height. Must be >= 1.
 879
 880    Returns:
 881        A 2D uint8 array of shape (height, width) with values 0 or 255.
 882
 883    Raises:
 884        ValueError: If count_x or count_y is less than 1.
 885    """
 886    if count_x < 1:
 887        raise ValueError(f"count_x {count_x} must be >= 1")
 888    if count_y < 1:
 889        raise ValueError(f"count_y {count_y} must be >= 1")
 890
 891    rows = (np.arange(height) * count_y // height) % 2
 892    cols = (np.arange(width) * count_x // width) % 2
 893    pattern = np.logical_xor(rows[:, None], cols[None, :])
 894    return (pattern * 255).astype(np.uint8)

 953def rgba_to_gray(arr: np.ndarray) -> np.ndarray:
 954    """Convert an RGB(A) image to grayscale by averaging the RGB channels.
 955
 956    Args:
 957        arr: Input image array, either 2D (grayscale) or 3D (color).
 958
 959    Returns:
 960        A 2D grayscale image array. If the input is already 2D, it is
 961        returned unchanged.
 962

 964        ValueError: If the array is neither 2D nor a 3-or-4-channel 3D array.
 965    """

 967        return arr

 969    if is_rgba(arr):
 970        return np.mean(arr[:, :, :3], axis=2).astype(arr.dtype)
 971
 972    raise ValueError(
 973        "Input array must be either 2D (grayscale) or 3D with 3 or 4 channels (color)."
 974    )

 977def combine(*, a: np.ndarray, b: np.ndarray) -> np.ndarray:
 978    """Combine two images by averaging their pixel values.
 979
 980    Args:
 981        a: First image, 2D grayscale or 3D color.
 982        b: Second image, same shape as `a` once converted to grayscale.
 983
 984    Returns:
 985        A 2D uint8 array, normalized to the range [0, 255].
 986
 987    Raises:
 988        ValueError: If the grayscale-converted images don't share a shape.
 989    """
 990    gray_a = rgba_to_gray(a)
 991    gray_b = rgba_to_gray(b)
 992

1002def brightness(*, arr: np.ndarray, factor: float) -> np.ndarray:
1003    """Adjust image brightness by an additive shift, clipped to [0, 255].
1004
1005    Brightness *translates* the pixel-intensity histogram: every pixel is
1006    shifted by the same amount, so dark areas lighten right along with
1007    bright ones (unlike a multiplicative scale, where a black pixel would
1008    stay exactly black). factor=1.0 leaves the image unchanged; factor=1.5
1009    shifts every pixel by +127.5*0.5 = +63.75, factor=2.0 by +127.5.
1010
1011    Args:
1012        arr: A 2D grayscale image array.
1013        factor: Brightness factor. 1.0 is unchanged; > 1.0 brightens
1014            (shifts toward white); < 1.0 darkens (shifts toward black).
1015
1016    Returns:
1017        A 2D uint8 array, same shape as `arr`.
1018    """
1019    max_pixel_value = 255.0
1020    offset = (factor - 1.0) * (max_pixel_value / 2.0)
1021    shifted = arr.astype(np.float64) + offset
1022    return np.clip(shifted, 0, max_pixel_value).astype(np.uint8)

1025def contrast(*, arr: np.ndarray, factor: float) -> np.ndarray:
1026    """Adjust image contrast by scaling around the mean, clipped to [0, 255].
1027
1028    Contrast *stretches* the pixel-intensity histogram outward from its own
1029    mean, rather than shifting it: the mean stays roughly the same, while
1030    values spread further from it. factor=1.0 leaves the image unchanged;
1031    factor=0.0 collapses every pixel to the mean (a flat gray image);
1032    factor > 1.0 pushes values further toward 0 and 255.
1033
1034    Args:
1035        arr: A 2D grayscale image array.
1036        factor: Contrast factor. 1.0 is unchanged; > 1.0 increases
1037            contrast; between 0.0 and 1.0 decreases it.
1038
1039    Returns:
1040        A 2D uint8 array, same shape as `arr`.
1041    """
1042    mean = arr.astype(np.float64).mean()
1043    stretched = (arr.astype(np.float64) - mean) * factor + mean
1044    return np.clip(stretched, 0, 255).astype(np.uint8)

1047def _backward_map(
1048    arr: np.ndarray, xs_source: np.ndarray, ys_source: np.ndarray
1049) -> np.ndarray:
1050    """Sample `arr` via bilinear interpolation at (xs_source, ys_source).
1051
1052    Shared by geometric-transform functions (`stretch`, `translate`, ...):
1053    each computes where every output pixel's source coordinate falls under
1054    its own inverse transform, then hands the resulting coordinate grids
1055    here to do the actual sampling. Coordinates outside `arr`'s bounds are
1056    filled with black (0).
1057
1058    Args:
1059        arr: A 2D grayscale image array.
1060        xs_source: Source x-coordinate for each output pixel, shape (height, width).
1061        ys_source: Source y-coordinate for each output pixel, shape (height, width).
1062
1063    Returns:
1064        A 2D uint8 array, same shape as `arr`.
1065    """
1066    height, width = arr.shape
1067    interpolator = RegularGridInterpolator(
1068        points=(np.arange(height), np.arange(width)),
1069        values=arr.astype(np.float64),
1070        method="linear",
1071        bounds_error=False,
1072        fill_value=0.0,
1073    )
1074    points = np.stack((ys_source.ravel(), xs_source.ravel()), axis=1)
1075    deformed = interpolator(points).reshape(arr.shape)
1076    return np.clip(deformed, 0, 255).astype(np.uint8)

1079def stretch(
1080    *, arr: np.ndarray, factor_x: float = 1.0, factor_y: float = 1.0
1081) -> np.ndarray:
1082    """Apply a uniaxial or biaxial stretch, pivoting on the image origin.
1083
1084    Mimics a continuum-mechanics stretch deformation gradient
1085    diag(factor_x, factor_y), anchored at the image's top-left corner
1086    (x=0, y=0): that corner stays fixed, and content grows (factor > 1.0)
1087    or shrinks (factor < 1.0) away from it along each axis. Uses backward
1088    mapping — for each output pixel, the inverse of the stretch locates
1089    its source coordinate in `arr`, with bilinear interpolation for
1090    non-integer source coordinates — so the result has no gaps, unlike
1091    naively moving each source pixel forward. A factor < 1.0 shrinks
1092    content toward the origin, leaving black (fill value 0) margins along
1093    the far (bottom/right) edges.
1094
1095    Args:
1096        arr: A 2D grayscale image array.
1097        factor_x: Stretch factor along the x-axis. Must be > 0.
1098        factor_y: Stretch factor along the y-axis. Must be > 0.
1099
1100    Returns:
1101        A 2D uint8 array, same shape as `arr`.
1102
1103    Raises:
1104        ValueError: If factor_x or factor_y is <= 0.
1105    """
1106    if factor_x <= 0:
1107        raise ValueError(f"factor_x {factor_x} must be > 0")
1108    if factor_y <= 0:
1109        raise ValueError(f"factor_y {factor_y} must be > 0")
1110
1111    height, width = arr.shape
1112    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
1113
1114    # Backward mapping, pivoting on the origin (0, 0): for each output
1115    # pixel, the inverse of the stretch gives the coordinate to sample
1116    # from in the original image.
1117    xs_source = xs / factor_x
1118    ys_source = ys / factor_y
1119
1120    return _backward_map(arr, xs_source, ys_source)

1123def translate(*, arr: np.ndarray, dx: float = 0.0, dy: float = 0.0) -> np.ndarray:
1124    """Apply a rigid-body translation: every pixel shifts by (dx, dy).
1125
1126    A pure displacement, with no change in shape or size — the simplest
1127    transformation category. Uses the same backward-mapping approach as
1128    `stretch`, so non-integer displacements are handled with bilinear
1129    interpolation rather than rounding. Content shifted in from outside
1130    the original bounds is filled with black (fill value 0).
1131
1132    Args:
1133        arr: A 2D grayscale image array.
1134        dx: Displacement in pixels along the x-axis; positive moves
1135            content right.
1136        dy: Displacement in pixels along the y-axis; positive moves
1137            content down.
1138
1139    Returns:
1140        A 2D uint8 array, same shape as `arr`.
1141    """
1142    height, width = arr.shape
1143    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
1144
1145    # Backward mapping: the source of output pixel (x, y) is (x - dx, y - dy).
1146    xs_source = xs - dx
1147    ys_source = ys - dy
1148
1149    return _backward_map(arr, xs_source, ys_source)

1152def rotate(*, arr: np.ndarray, angle: float) -> np.ndarray:
1153    """Apply a rigid-body rotation, pivoting on the image origin.
1154
1155    Rotates content by `angle` degrees, positive counterclockwise,
1156    pivoting on the image's top-left corner (0, 0) rather than its
1157    center — consistent with `stretch` and `translate`'s pivot choice in
1158    this codebase, but unlike a typical "object spins in place" rotation
1159    example: most content swings away from that fixed corner, similar to
1160    a door on a hinge. Uses the same backward-mapping approach as
1161    `stretch` and `translate`, so non-integer source coordinates are
1162    bilinearly interpolated, and any pixel with no corresponding source
1163    coordinate within `arr`'s bounds is filled with black (fill value 0).
1164
1165    Args:
1166        arr: A 2D grayscale image array.
1167        angle: Rotation angle in degrees; positive is counterclockwise.
1168
1169    Returns:
1170        A 2D uint8 array, same shape as `arr`.
1171    """
1172    height, width = arr.shape
1173    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
1174
1175    theta = np.deg2rad(angle)
1176    cos_theta = np.cos(theta)
1177    sin_theta = np.sin(theta)
1178
1179    # Backward mapping: applying the inverse (-angle) rotation to each
1180    # output coordinate gives the coordinate to sample from.
1181    xs_source = cos_theta * xs + sin_theta * ys
1182    ys_source = -sin_theta * xs + cos_theta * ys
1183
1184    return _backward_map(arr, xs_source, ys_source)

1187def shear(*, arr: np.ndarray, shear_x: float = 0.0, shear_y: float = 0.0) -> np.ndarray:
1188    """Apply a simple shear, pivoting on the image origin.
1189
1190    Mimics a continuum-mechanics shear deformation gradient
1191    [[1, shear_x], [shear_y, 1]], anchored at the image's top-left corner
1192    (x=0, y=0), consistent with `stretch`, `translate`, and `rotate`'s
1193    pivot choice in this codebase: horizontal planes slide relative to
1194    each other by an amount proportional to their y-coordinate
1195    (shear_x), and/or vertical planes slide by an amount proportional to
1196    their x-coordinate (shear_y). Uses the same backward-mapping approach
1197    as the other transform functions, so non-integer source coordinates
1198    are bilinearly interpolated, and any pixel with no corresponding
1199    source coordinate within `arr`'s bounds is filled with black (fill
1200    value 0).
1201
1202    Args:
1203        arr: A 2D grayscale image array.
1204        shear_x: Horizontal shear factor.
1205        shear_y: Vertical shear factor.
1206
1207    Returns:
1208        A 2D uint8 array, same shape as `arr`.
1209
1210    Raises:
1211        ValueError: If shear_x * shear_y == 1, which makes the
1212            deformation gradient singular (non-invertible).
1213    """
1214    determinant = 1.0 - shear_x * shear_y
1215    if determinant == 0:
1216        raise ValueError(
1217            f"shear_x {shear_x} and shear_y {shear_y} produce a singular "
1218            "deformation gradient (shear_x * shear_y == 1)"
1219        )
1220
1221    height, width = arr.shape
1222    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
1223
1224    # Backward mapping: invert the shear deformation gradient
1225    # [[1, shear_x], [shear_y, 1]] to find each output pixel's source.
1226    xs_source = (xs - shear_x * ys) / determinant
1227    ys_source = (-shear_y * xs + ys) / determinant
1228
1229    return _backward_map(arr, xs_source, ys_source)

1232def complex_deform(
1233    *,
1234    arr: np.ndarray,
1235    factor_x: float = 1.0,
1236    factor_y: float = 1.0,
1237    angle: float = 0.0,
1238) -> np.ndarray:
1239    """Apply an anisotropic stretch composed with a rotation, in one pass.
1240
1241    Composes a stretch (deformation gradient diag(factor_x, factor_y))
1242    with a rotation (`angle` degrees, counterclockwise): the combined
1243    deformation gradient is F = R(angle) @ diag(factor_x, factor_y), i.e.
1244    the stretch is applied first and the rotation second. Applying both
1245    in a single backward-mapping pass (rather than calling `stretch` and
1246    then `rotate` separately) avoids the extra blur of interpolating
1247    twice. Represents realistic loading scenarios where materials
1248    experience multiple simultaneous deformation modes — typically the
1249    hardest case for correlation algorithms. Pivots on the image's
1250    top-left corner (0, 0), consistent with `stretch`, `translate`,
1251    `rotate`, and `shear`'s pivot choice in this codebase.
1252
1253    Args:
1254        arr: A 2D grayscale image array.
1255        factor_x: Stretch factor along the x-axis, applied before the
1256            rotation. Must be > 0.
1257        factor_y: Stretch factor along the y-axis, applied before the
1258            rotation. Must be > 0.
1259        angle: Rotation angle in degrees, applied after the stretch;
1260            positive is counterclockwise.
1261
1262    Returns:
1263        A 2D uint8 array, same shape as `arr`.
1264
1265    Raises:
1266        ValueError: If factor_x or factor_y is <= 0.
1267    """
1268    if factor_x <= 0:
1269        raise ValueError(f"factor_x {factor_x} must be > 0")
1270    if factor_y <= 0:
1271        raise ValueError(f"factor_y {factor_y} must be > 0")
1272
1273    height, width = arr.shape
1274    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
1275
1276    theta = np.deg2rad(angle)
1277    cos_theta = np.cos(theta)
1278    sin_theta = np.sin(theta)
1279
1280    # Backward mapping for F = R(angle) @ diag(factor_x, factor_y):
1281    # F_inv = diag(1/factor_x, 1/factor_y) @ R(-angle), applied to each
1282    # output coordinate to find its source. Un-rotate first, then unscale.
1283    xs_rotated = cos_theta * xs + sin_theta * ys
1284    ys_rotated = -sin_theta * xs + cos_theta * ys
1285    xs_source = xs_rotated / factor_x
1286    ys_source = ys_rotated / factor_y
1287
1288    return _backward_map(arr, xs_source, ys_source)

1291def crack_dislocation(*, arr: np.ndarray, offset: float = 8.0) -> np.ndarray:
1292    """Apply a discontinuous vertical-crack displacement field.
1293
1294    Splits the image with a vertical crack at x = width / 2: the left
1295    half shifts down by `offset` pixels and the right half shifts up by
1296    `offset` pixels, producing a displacement field that jumps
1297    discontinuously across the crack line — unlike every other transform
1298    in this module, which varies smoothly. Standard DIC assumes smooth
1299    displacements and cannot capture this jump; cases like this motivate
1300    the Heaviside finite-element formulation. Uses the same backward
1301    mapping as the other transform functions, just with a piecewise
1302    (rather than single-matrix) displacement field, so non-integer
1303    source coordinates are bilinearly interpolated, and any pixel with
1304    no corresponding source coordinate within `arr`'s bounds is filled
1305    with black (fill value 0).
1306
1307    Args:
1308        arr: A 2D grayscale image array.
1309        offset: Displacement in pixels; the left half (x < width / 2)
1310            shifts down by this amount, the right half shifts up by it.
1311
1312    Returns:
1313        A 2D uint8 array, same shape as `arr`.
1314    """
1315    height, width = arr.shape
1316    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
1317
1318    xs_source = xs.astype(np.float64)
1319    ys_source = ys.astype(np.float64)
1320
1321    left_half = xs < (width / 2)
1322    ys_source[left_half] -= offset
1323    ys_source[~left_half] += offset
1324
1325    return _backward_map(arr, xs_source, ys_source)

1341    """Write a NumPy array to an SVG file.

1361        f'  <image width="{width}" height="{height}" '
1362        f'xlink:href="data:image/png;base64,{encoded}"/>\n'

1884def _reticle_marker_path(
1885    *,
1886    ring_radius: float = 0.85,
1887    tick_outer: float = 1.20,
1888    num_circle_points: int = 64,
1889) -> MarkerPath:
1890    """A target-reticle glyph: a ring with four tick marks poking through it.
1891
1892    Built as a matplotlib marker path (unit-scaled to roughly [-1, 1]) meant
1893    to be *stroked*, not filled -- a circle outline, plus four short line
1894    segments along +/-x and +/-y running from the circle's own radius out
1895    past it, so the center stays fully open. Unlike a filled ring (an
1896    annulus), a stroked outline's thickness is set directly via
1897    `markeredgewidth` (in points), independent of this path's own geometry
1898    -- letting the line be pushed far thinner than a filled band can go
1899    before anti-aliasing makes it look patchy.
1900
1901    Args:
1902        ring_radius: The circle's radius; also each tick's start radius.
1903        tick_outer: Each tick's end radius (past `ring_radius`).
1904        num_circle_points: Number of vertices approximating the circle.
1905
1906    Returns:
1907        A compound `matplotlib.path.Path` usable as a `marker=` argument,
1908        with `markerfacecolor="none"` so only its stroke renders.
1909    """
1910    theta = np.linspace(0, 2 * np.pi, num_circle_points, endpoint=False)
1911    circle = np.column_stack((ring_radius * np.cos(theta), ring_radius * np.sin(theta)))
1912    # Path(..., closed=True) treats the *last* vertex as the ignored
1913    # CLOSEPOLY placeholder rather than drawing through it, so it must be a
1914    # repeat of the first vertex -- otherwise the final edge is dropped.
1915    circle_path = MarkerPath(np.vstack([circle, circle[:1]]), closed=True)
1916
1917    tick_codes = [MarkerPath.MOVETO, MarkerPath.LINETO]
1918    tick_paths = [
1919        MarkerPath(
1920            [
1921                (direction_x * ring_radius, direction_y * ring_radius),
1922                (direction_x * tick_outer, direction_y * tick_outer),
1923            ],
1924            codes=tick_codes,
1925        )
1926        for direction_x, direction_y in [(1, 0), (-1, 0), (0, 1), (0, -1)]
1927    ]
1928
1929    return MarkerPath.make_compound_path(circle_path, *tick_paths)

 64    *, image: np.ndarray, origin: PixelCoordinate, width: int, height: int
 65) -> np.ndarray:
 66    """Extract a width x height crop of `image` with its top-left corner at `origin`.
 67

 69    top-left corner is `(0, 0)`) and may lie partially or entirely outside
 70    `image`'s bounds. Wherever the requested region doesn't overlap
 71    `image`, the corresponding output pixels are zero (black) rather than
 72    raising an error — so a subimage straddling an edge, or lying
 73    completely outside `image`, is always a valid, well-shaped result.
 74
 75    Args:
 76        image: Source 2D grayscale image array.
 77        origin: Top-left corner of the region to extract, in `image`'s
 78            pixel reference frame.
 79        width: Width of the extracted region in pixels. Must be >= 1.
 80        height: Height of the extracted region in pixels. Must be >= 1.
 81
 82    Returns:
 83        A 2D array of shape (height, width), same dtype as `image`.
 84
 85    Raises:
 86        ValueError: If `image` is not 2D, or width or height is less than 1.
 87    """
 88    if image.ndim != 2:
 89        raise ValueError(f"image must be 2D, got shape {image.shape}")
 90    if width < 1:
 91        raise ValueError(f"width {width} must be >= 1")
 92    if height < 1:
 93        raise ValueError(f"height {height} must be >= 1")
 94
 95    image_height, image_width = image.shape
 96    result = np.zeros((height, width), dtype=image.dtype)
 97
 98    src_x_start = max(origin.x, 0)
 99    src_y_start = max(origin.y, 0)
100    src_x_end = min(origin.x + width, image_width)
101    src_y_end = min(origin.y + height, image_height)
102
103    if src_x_start >= src_x_end or src_y_start >= src_y_end:
104        return result  # requested region doesn't overlap image at all
105
106    dst_x_start = src_x_start - origin.x
107    dst_y_start = src_y_start - origin.y
108    dst_x_end = dst_x_start + (src_x_end - src_x_start)
109    dst_y_end = dst_y_start + (src_y_end - src_y_start)
110
111    result[dst_y_start:dst_y_end, dst_x_start:dst_x_end] = image[
112        src_y_start:src_y_end, src_x_start:src_x_end
113    ]
114    return result
