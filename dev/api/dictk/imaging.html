 16class PixelCoordinate(NamedTuple):
 17    """A pixel location in an image's reference frame.
 18
 19    Attributes:
 20        x: Horizontal pixel coordinate, increasing left to right.
 21        y: Vertical pixel coordinate, increasing top to bottom.

 25    y: int

 37def subimage(
 38    image: np.ndarray, origin: PixelCoordinate, width: int, height: int
 39) -> np.ndarray:
 40    """Extract a width x height crop of `image` with its top-left corner at `origin`.
 41
 42    `origin` is expressed in `image`'s own pixel reference frame (`image`'s
 43    top-left corner is `(0, 0)`) and may lie partially or entirely outside
 44    `image`'s bounds. Wherever the requested region doesn't overlap
 45    `image`, the corresponding output pixels are zero (black) rather than
 46    raising an error — so a subimage straddling an edge, or lying
 47    completely outside `image`, is always a valid, well-shaped result.
 48
 49    Args:
 50        image: Source 2D grayscale image array.
 51        origin: Top-left corner of the region to extract, in `image`'s
 52            pixel reference frame.
 53        width: Width of the extracted region in pixels. Must be >= 1.
 54        height: Height of the extracted region in pixels. Must be >= 1.
 55
 56    Returns:
 57        A 2D array of shape (height, width), same dtype as `image`.
 58
 59    Raises:
 60        ValueError: If width or height is less than 1.
 61    """
 62    if width < 1:
 63        raise ValueError(f"width {width} must be >= 1")
 64    if height < 1:
 65        raise ValueError(f"height {height} must be >= 1")
 66
 67    image_height, image_width = image.shape
 68    result = np.zeros((height, width), dtype=image.dtype)
 69
 70    src_x_start = max(origin.x, 0)
 71    src_y_start = max(origin.y, 0)
 72    src_x_end = min(origin.x + width, image_width)
 73    src_y_end = min(origin.y + height, image_height)
 74
 75    if src_x_start >= src_x_end or src_y_start >= src_y_end:
 76        return result  # requested region doesn't overlap image at all
 77
 78    dst_x_start = src_x_start - origin.x
 79    dst_y_start = src_y_start - origin.y
 80    dst_x_end = dst_x_start + (src_x_end - src_x_start)
 81    dst_y_end = dst_y_start + (src_y_end - src_y_start)
 82
 83    result[dst_y_start:dst_y_end, dst_x_start:dst_x_end] = image[
 84        src_y_start:src_y_end, src_x_start:src_x_end
 85    ]
 86    return result

 89def plot_subimage_bounds(
 90    image: np.ndarray,
 91    origin: PixelCoordinate,
 92    width: int,
 93    height: int,
 94    path: Path,
 95    dpi: int = 300,
 96) -> None:
 97    """Save a figure overlaying a source image's bounds and a subimage region.
 98
 99    Draws `image`'s own bounds in blue and the requested `origin`/`width`/
100    `height` region in red, on top of `image` itself — including cases
101    where the red region extends partially or completely outside the blue
102    one, to visualize what `subimage()` would crop from. An `'o'` marker
103    is also drawn at each rectangle's own origin: blue at `(0, 0)` for
104    `image`, red at `origin` for the subimage — both in `image`'s pixel
105    reference frame.
106
107    Args:
108        image: Source 2D grayscale image array.
109        origin: Top-left corner of the region, in `image`'s pixel
110            reference frame; see `subimage()`.
111        width: Width of the region in pixels. Must be >= 1.
112        height: Height of the region in pixels. Must be >= 1.
113        path: Output file path for the figure; format is inferred from
114            the extension by matplotlib's savefig (e.g. .png), not
115            dictk's own write_image/write_svg.
116        dpi: Resolution of the saved figure.
117
118    Raises:
119        ValueError: If width or height is less than 1.
120    """
121    if width < 1:
122        raise ValueError(f"width {width} must be >= 1")
123    if height < 1:
124        raise ValueError(f"height {height} must be >= 1")
125
126    image_height, image_width = image.shape
127
128    margin = max(width, height, image_width, image_height) * 0.05
129    x_min = min(0, origin.x) - margin
130    x_max = max(image_width, origin.x + width) + margin
131    y_min = min(0, origin.y) - margin
132    y_max = max(image_height, origin.y + height) + margin
133
134    figsize = (
135        (x_max - x_min) / _FIGURE_PIXELS_PER_INCH,
136        (y_max - y_min) / _FIGURE_PIXELS_PER_INCH,
137    )
138    fig, ax = plt.subplots(figsize=figsize)
139    ax.imshow(
140        image, cmap="gray", origin="upper", extent=(0, image_width, image_height, 0)
141    )
142
143    ax.add_patch(
144        patches.Rectangle(
145            (0, 0),
146            image_width,
147            image_height,
148            edgecolor="blue",
149            facecolor="none",
150            linewidth=1.5,
151            label="source image",
152        )
153    )
154    ax.add_patch(
155        patches.Rectangle(
156            (origin.x, origin.y),
157            width,
158            height,
159            edgecolor="red",
160            facecolor="none",
161            linewidth=1.5,
162            label="subimage",
163        )
164    )
165    ax.plot(0, 0, marker="o", color="blue", markersize=6)
166    ax.plot(origin.x, origin.y, marker="o", color="red", markersize=6)
167
168    ax.set_xlim(x_min, x_max)
169    ax.set_ylim(y_max, y_min)  # inverted: image y increases downward
170
171    ax.set_xlabel("x (pixels)")
172    ax.set_ylabel("y (pixels)")
173    # Legend placed fully outside the axes (not just an "upper right"-style
174    # corner) so it never overlaps the source or subimage rectangles,
175    # regardless of how much of the frame they fill; bbox_inches="tight"
176    # on save keeps it from being clipped off the saved figure.
177    ax.legend(loc="center left", bbox_to_anchor=(1.02, 0.5), fontsize=8)
178    ax.set_title(f"subimage bounds: origin=({origin.x}, {origin.y}), {width}x{height}")
179
180    plt.savefig(path, dpi=dpi, bbox_inches="tight")
181    plt.close(fig)

185    image: np.ndarray,
186    origin: PixelCoordinate,
187    width: int,
188    height: int,
189    path: Path,
190    dpi: int = 300,
191) -> None:
192    """Save a figure of the subimage itself, in its own local reference frame.
193
194    Extracts the `width` x `height` region at `origin` (via `subimage()`)

246    ax.set_xlim(x_min, x_max)
247    ax.set_ylim(y_max, y_min)  # inverted: image y increases downward

366def checkerboard(
367    width: int, height: int, count_x: int = 8, count_y: int = 8
368) -> np.ndarray:
369    """Generate a black-and-white checkerboard test image.
370
371    Same parameters and pixel values as the `dictk checkerboard` CLI
372    command, minus the file write — see the `dictk` package docstring for
373    why the API layer stops at the array. count_x and count_y are named for
374    the rectangles they divide the image into, not "squares": unequal
375    counts (or a width/height ratio that doesn't match count_x/count_y)
376    produce rectangular cells, not square ones.
377
378    Args:
379        width: Image width in pixels.
380        height: Image height in pixels.
381        count_x: Number of rectangles along the width. Must be >= 1.
382        count_y: Number of rectangles along the height. Must be >= 1.
383
384    Returns:
385        A 2D uint8 array of shape (height, width) with values 0 or 255.
386
387    Raises:
388        ValueError: If count_x or count_y is less than 1.
389    """
390    if count_x < 1:
391        raise ValueError(f"count_x {count_x} must be >= 1")
392    if count_y < 1:
393        raise ValueError(f"count_y {count_y} must be >= 1")
394
395    rows = (np.arange(height) * count_y // height) % 2
396    cols = (np.arange(width) * count_x // width) % 2
397    pattern = np.logical_xor(rows[:, None], cols[None, :])
398    return (pattern * 255).astype(np.uint8)

401def astronaut(width: int = 512, height: int = 512) -> np.ndarray:
402    """Load a bundled real-world grayscale reference image.
403
404    Same parameters and pixel values as the `dictk astronaut` CLI
405    command, minus the file write — see the `dictk` package docstring for
406    why the API layer stops at the array.
407
408    The source is a NASA portrait of astronaut Eileen Collins, from the
409    NASA Great Images database ("No known copyright restrictions, released
410    into the public domain."). It's bundled as a color image and converted
411    with `rgba_to_gray`; unlike `rosta` and `checkerboard`, it isn't
412    procedurally generated, so `width`/`height` other than the native
413    512x512 resize the source image (via `scipy.ndimage.zoom`) rather than
414    computing a fresh pattern at that size.
415
416    Args:
417        width: Image width in pixels. Must be >= 1.
418        height: Image height in pixels. Must be >= 1.
419
420    Returns:
421        A 2D uint8 array of shape (height, width).
422
423    Raises:
424        ValueError: If width or height is less than 1.
425    """
426    if width < 1:
427        raise ValueError(f"width {width} must be >= 1")
428    if height < 1:
429        raise ValueError(f"height {height} must be >= 1")
430
431    asset_path = importlib.resources.files("dictk") / "data" / "astronaut.png"
432    with importlib.resources.as_file(asset_path) as path:
433        color = read_image(path)
434    gray = rgba_to_gray(color)
435
436    native_height, native_width = gray.shape
437    if (width, height) == (native_width, native_height):
438        return gray
439
440    zoom_factors = (height / native_height, width / native_width)
441    resized = zoom(gray.astype(np.float64), zoom_factors, order=3)
442    return np.clip(resized, 0, 255).astype(np.uint8)

457def rgba_to_gray(arr: np.ndarray) -> np.ndarray:
458    """Convert an RGB(A) image to grayscale by averaging the RGB channels.
459
460    Args:
461        arr: Input image array, either 2D (grayscale) or 3D (color).
462
463    Returns:
464        A 2D grayscale image array. If the input is already 2D, it is
465        returned unchanged.
466
467    Raises:
468        ValueError: If the array is neither 2D nor a 3-or-4-channel 3D array.
469    """
470    if arr.ndim == 2:
471        return arr
472
473    if is_rgba(arr):
474        return np.mean(arr[:, :, :3], axis=2).astype(arr.dtype)
475
476    raise ValueError(
477        "Input array must be either 2D (grayscale) or 3D with 3 or 4 channels (color)."
478    )

481def combine_images(a: np.ndarray, b: np.ndarray) -> np.ndarray:
482    """Combine two images by averaging their pixel values.
483
484    Args:
485        a: First image, 2D grayscale or 3D color.
486        b: Second image, same shape as `a` once converted to grayscale.
487
488    Returns:
489        A 2D uint8 array, normalized to the range [0, 255].
490
491    Raises:
492        ValueError: If the grayscale-converted images don't share a shape.
493    """
494    gray_a = rgba_to_gray(a)
495    gray_b = rgba_to_gray(b)
496
497    if gray_a.shape != gray_b.shape:
498        raise ValueError(
499            f"shape mismatch: a.shape={gray_a.shape}, b.shape={gray_b.shape}"
500        )
501
502    combined = gray_a.astype(np.float64) + gray_b.astype(np.float64)
503    return (combined / combined.max() * 255).astype(np.uint8)

506def brightness(arr: np.ndarray, factor: float) -> np.ndarray:
507    """Adjust image brightness by an additive shift, clipped to [0, 255].
508
509    Brightness *translates* the pixel-intensity histogram: every pixel is
510    shifted by the same amount, so dark areas lighten right along with
511    bright ones (unlike a multiplicative scale, where a black pixel would
512    stay exactly black). factor=1.0 leaves the image unchanged; factor=1.5
513    shifts every pixel by +127.5*0.5 = +63.75, factor=2.0 by +127.5.
514
515    Args:
516        arr: A 2D grayscale image array.
517        factor: Brightness factor. 1.0 is unchanged; > 1.0 brightens
518            (shifts toward white); < 1.0 darkens (shifts toward black).
519
520    Returns:
521        A 2D uint8 array, same shape as `arr`.
522    """
523    max_pixel_value = 255.0
524    offset = (factor - 1.0) * (max_pixel_value / 2.0)
525    shifted = arr.astype(np.float64) + offset
526    return np.clip(shifted, 0, max_pixel_value).astype(np.uint8)

529def contrast(arr: np.ndarray, factor: float) -> np.ndarray:
530    """Adjust image contrast by scaling around the mean, clipped to [0, 255].
531
532    Contrast *stretches* the pixel-intensity histogram outward from its own
533    mean, rather than shifting it: the mean stays roughly the same, while
534    values spread further from it. factor=1.0 leaves the image unchanged;
535    factor=0.0 collapses every pixel to the mean (a flat gray image);
536    factor > 1.0 pushes values further toward 0 and 255.
537
538    Args:
539        arr: A 2D grayscale image array.
540        factor: Contrast factor. 1.0 is unchanged; > 1.0 increases
541            contrast; between 0.0 and 1.0 decreases it.
542
543    Returns:
544        A 2D uint8 array, same shape as `arr`.
545    """
546    mean = arr.astype(np.float64).mean()
547    stretched = (arr.astype(np.float64) - mean) * factor + mean
548    return np.clip(stretched, 0, 255).astype(np.uint8)

551def _backward_map(
552    arr: np.ndarray, xs_source: np.ndarray, ys_source: np.ndarray
553) -> np.ndarray:
554    """Sample `arr` via bilinear interpolation at (xs_source, ys_source).
555
556    Shared by geometric-transform functions (`stretch`, `translate`, ...):
557    each computes where every output pixel's source coordinate falls under
558    its own inverse transform, then hands the resulting coordinate grids
559    here to do the actual sampling. Coordinates outside `arr`'s bounds are
560    filled with black (0).
561
562    Args:
563        arr: A 2D grayscale image array.
564        xs_source: Source x-coordinate for each output pixel, shape (height, width).
565        ys_source: Source y-coordinate for each output pixel, shape (height, width).
566
567    Returns:
568        A 2D uint8 array, same shape as `arr`.
569    """
570    height, width = arr.shape
571    interpolator = RegularGridInterpolator(
572        points=(np.arange(height), np.arange(width)),
573        values=arr.astype(np.float64),
574        method="linear",
575        bounds_error=False,
576        fill_value=0.0,
577    )
578    points = np.stack((ys_source.ravel(), xs_source.ravel()), axis=1)
579    deformed = interpolator(points).reshape(arr.shape)
580    return np.clip(deformed, 0, 255).astype(np.uint8)

583def stretch(
584    arr: np.ndarray, factor_x: float = 1.0, factor_y: float = 1.0
585) -> np.ndarray:
586    """Apply a uniaxial or biaxial stretch, pivoting on the image origin.
587
588    Mimics a continuum-mechanics stretch deformation gradient
589    diag(factor_x, factor_y), anchored at the image's top-left corner
590    (x=0, y=0): that corner stays fixed, and content grows (factor > 1.0)
591    or shrinks (factor < 1.0) away from it along each axis. Uses backward
592    mapping — for each output pixel, the inverse of the stretch locates
593    its source coordinate in `arr`, with bilinear interpolation for
594    non-integer source coordinates — so the result has no gaps, unlike
595    naively moving each source pixel forward. A factor < 1.0 shrinks
596    content toward the origin, leaving black (fill value 0) margins along
597    the far (bottom/right) edges.
598
599    Args:
600        arr: A 2D grayscale image array.
601        factor_x: Stretch factor along the x-axis. Must be > 0.
602        factor_y: Stretch factor along the y-axis. Must be > 0.
603
604    Returns:
605        A 2D uint8 array, same shape as `arr`.
606
607    Raises:
608        ValueError: If factor_x or factor_y is <= 0.
609    """
610    if factor_x <= 0:
611        raise ValueError(f"factor_x {factor_x} must be > 0")
612    if factor_y <= 0:
613        raise ValueError(f"factor_y {factor_y} must be > 0")
614
615    height, width = arr.shape
616    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
617
618    # Backward mapping, pivoting on the origin (0, 0): for each output
619    # pixel, the inverse of the stretch gives the coordinate to sample
620    # from in the original image.
621    xs_source = xs / factor_x
622    ys_source = ys / factor_y
623
624    return _backward_map(arr, xs_source, ys_source)

627def translate(arr: np.ndarray, dx: float = 0.0, dy: float = 0.0) -> np.ndarray:
628    """Apply a rigid-body translation: every pixel shifts by (dx, dy).
629
630    A pure displacement, with no change in shape or size — the simplest
631    transformation category. Uses the same backward-mapping approach as
632    `stretch`, so non-integer displacements are handled with bilinear
633    interpolation rather than rounding. Content shifted in from outside
634    the original bounds is filled with black (fill value 0).
635
636    Args:
637        arr: A 2D grayscale image array.

642

645    """

647    xs, ys = np.meshgrid(np.arange(width), np.arange(height))

649    # Backward mapping: the source of output pixel (x, y) is (x - dx, y - dy).

651    ys_source = ys - dy

656def rotate(arr: np.ndarray, angle: float) -> np.ndarray:
657    """Apply a rigid-body rotation, pivoting on the image origin.
658
659    Rotates content by `angle` degrees, positive counterclockwise,
660    pivoting on the image's top-left corner (0, 0) rather than its
661    center — consistent with `stretch` and `translate`'s pivot choice in
662    this codebase, but unlike a typical "object spins in place" rotation
663    example: most content swings away from that fixed corner, similar to
664    a door on a hinge. Uses the same backward-mapping approach as
665    `stretch` and `translate`, so non-integer source coordinates are
666    bilinearly interpolated, and any pixel with no corresponding source
667    coordinate within `arr`'s bounds is filled with black (fill value 0).
668
669    Args:
670        arr: A 2D grayscale image array.
671        angle: Rotation angle in degrees; positive is counterclockwise.
672
673    Returns:
674        A 2D uint8 array, same shape as `arr`.
675    """
676    height, width = arr.shape
677    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
678
679    theta = np.deg2rad(angle)
680    cos_theta = np.cos(theta)
681    sin_theta = np.sin(theta)
682
683    # Backward mapping: applying the inverse (-angle) rotation to each
684    # output coordinate gives the coordinate to sample from.
685    xs_source = cos_theta * xs + sin_theta * ys
686    ys_source = -sin_theta * xs + cos_theta * ys
687
688    return _backward_map(arr, xs_source, ys_source)

691def shear(arr: np.ndarray, shear_x: float = 0.0, shear_y: float = 0.0) -> np.ndarray:
692    """Apply a simple shear, pivoting on the image origin.
693
694    Mimics a continuum-mechanics shear deformation gradient
695    [[1, shear_x], [shear_y, 1]], anchored at the image's top-left corner
696    (x=0, y=0), consistent with `stretch`, `translate`, and `rotate`'s
697    pivot choice in this codebase: horizontal planes slide relative to
698    each other by an amount proportional to their y-coordinate
699    (shear_x), and/or vertical planes slide by an amount proportional to
700    their x-coordinate (shear_y). Uses the same backward-mapping approach
701    as the other transform functions, so non-integer source coordinates
702    are bilinearly interpolated, and any pixel with no corresponding
703    source coordinate within `arr`'s bounds is filled with black (fill
704    value 0).
705
706    Args:
707        arr: A 2D grayscale image array.
708        shear_x: Horizontal shear factor.
709        shear_y: Vertical shear factor.
710
711    Returns:
712        A 2D uint8 array, same shape as `arr`.
713
714    Raises:
715        ValueError: If shear_x * shear_y == 1, which makes the
716            deformation gradient singular (non-invertible).
717    """
718    determinant = 1.0 - shear_x * shear_y
719    if determinant == 0:
720        raise ValueError(
721            f"shear_x {shear_x} and shear_y {shear_y} produce a singular "
722            "deformation gradient (shear_x * shear_y == 1)"
723        )
724
725    height, width = arr.shape
726    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
727
728    # Backward mapping: invert the shear deformation gradient
729    # [[1, shear_x], [shear_y, 1]] to find each output pixel's source.
730    xs_source = (xs - shear_x * ys) / determinant
731    ys_source = (-shear_y * xs + ys) / determinant
732
733    return _backward_map(arr, xs_source, ys_source)

736def complex_deform(
737    arr: np.ndarray, factor_x: float = 1.0, factor_y: float = 1.0, angle: float = 0.0
738) -> np.ndarray:
739    """Apply an anisotropic stretch composed with a rotation, in one pass.
740
741    Composes a stretch (deformation gradient diag(factor_x, factor_y))
742    with a rotation (`angle` degrees, counterclockwise): the combined
743    deformation gradient is F = R(angle) @ diag(factor_x, factor_y), i.e.
744    the stretch is applied first and the rotation second. Applying both
745    in a single backward-mapping pass (rather than calling `stretch` and
746    then `rotate` separately) avoids the extra blur of interpolating
747    twice. Represents realistic loading scenarios where materials
748    experience multiple simultaneous deformation modes — typically the
749    hardest case for correlation algorithms. Pivots on the image's
750    top-left corner (0, 0), consistent with `stretch`, `translate`,
751    `rotate`, and `shear`'s pivot choice in this codebase.
752
753    Args:
754        arr: A 2D grayscale image array.
755        factor_x: Stretch factor along the x-axis, applied before the
756            rotation. Must be > 0.
757        factor_y: Stretch factor along the y-axis, applied before the
758            rotation. Must be > 0.
759        angle: Rotation angle in degrees, applied after the stretch;
760            positive is counterclockwise.
761
762    Returns:
763        A 2D uint8 array, same shape as `arr`.
764
765    Raises:
766        ValueError: If factor_x or factor_y is <= 0.
767    """
768    if factor_x <= 0:
769        raise ValueError(f"factor_x {factor_x} must be > 0")
770    if factor_y <= 0:
771        raise ValueError(f"factor_y {factor_y} must be > 0")
772
773    height, width = arr.shape
774    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
775
776    theta = np.deg2rad(angle)
777    cos_theta = np.cos(theta)
778    sin_theta = np.sin(theta)
779
780    # Backward mapping for F = R(angle) @ diag(factor_x, factor_y):
781    # F_inv = diag(1/factor_x, 1/factor_y) @ R(-angle), applied to each
782    # output coordinate to find its source. Un-rotate first, then unscale.
783    xs_rotated = cos_theta * xs + sin_theta * ys
784    ys_rotated = -sin_theta * xs + cos_theta * ys
785    xs_source = xs_rotated / factor_x
786    ys_source = ys_rotated / factor_y
787
788    return _backward_map(arr, xs_source, ys_source)

791def crack_dislocation(arr: np.ndarray, offset: float = 8.0) -> np.ndarray:
792    """Apply a discontinuous vertical-crack displacement field.
793
794    Splits the image with a vertical crack at x = width / 2: the left
795    half shifts down by `offset` pixels and the right half shifts up by
796    `offset` pixels, producing a displacement field that jumps
797    discontinuously across the crack line — unlike every other transform
798    in this module, which varies smoothly. Standard DIC assumes smooth
799    displacements and cannot capture this jump; cases like this motivate
800    the Heaviside finite-element formulation. Uses the same backward
801    mapping as the other transform functions, just with a piecewise
802    (rather than single-matrix) displacement field, so non-integer
803    source coordinates are bilinearly interpolated, and any pixel with
804    no corresponding source coordinate within `arr`'s bounds is filled
805    with black (fill value 0).
806
807    Args:
808        arr: A 2D grayscale image array.
809        offset: Displacement in pixels; the left half (x < width / 2)
810            shifts down by this amount, the right half shifts up by it.
811
812    Returns:
813        A 2D uint8 array, same shape as `arr`.
814    """
815    height, width = arr.shape
816    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
817
818    xs_source = xs.astype(np.float64)
819    ys_source = ys.astype(np.float64)
820
821    left_half = xs < (width / 2)
822    ys_source[left_half] -= offset
823    ys_source[~left_half] += offset
824
825    return _backward_map(arr, xs_source, ys_source)

38def subimage(
39    image: np.ndarray, origin: PixelCoordinate, width: int, height: int
40) -> np.ndarray:
41    """Extract a width x height crop of `image` with its top-left corner at `origin`.
42
43    `origin` is expressed in `image`'s own pixel reference frame (`image`'s
44    top-left corner is `(0, 0)`) and may lie partially or entirely outside
45    `image`'s bounds. Wherever the requested region doesn't overlap
46    `image`, the corresponding output pixels are zero (black) rather than
47    raising an error — so a subimage straddling an edge, or lying
48    completely outside `image`, is always a valid, well-shaped result.
49
50    Args:
51        image: Source 2D grayscale image array.
52        origin: Top-left corner of the region to extract, in `image`'s
53            pixel reference frame.
54        width: Width of the extracted region in pixels. Must be >= 1.
55        height: Height of the extracted region in pixels. Must be >= 1.
56
57    Returns:
58        A 2D array of shape (height, width), same dtype as `image`.
59
60    Raises:
61        ValueError: If width or height is less than 1.
62    """
63    if width < 1:
64        raise ValueError(f"width {width} must be >= 1")
65    if height < 1:
66        raise ValueError(f"height {height} must be >= 1")
67
68    image_height, image_width = image.shape
69    result = np.zeros((height, width), dtype=image.dtype)
70
71    src_x_start = max(origin.x, 0)
72    src_y_start = max(origin.y, 0)
73    src_x_end = min(origin.x + width, image_width)
74    src_y_end = min(origin.y + height, image_height)
75
76    if src_x_start >= src_x_end or src_y_start >= src_y_end:
77        return result  # requested region doesn't overlap image at all
78
79    dst_x_start = src_x_start - origin.x
80    dst_y_start = src_y_start - origin.y
81    dst_x_end = dst_x_start + (src_x_end - src_x_start)
82    dst_y_end = dst_y_start + (src_y_end - src_y_start)
83
84    result[dst_y_start:dst_y_end, dst_x_start:dst_x_end] = image[
85        src_y_start:src_y_end, src_x_start:src_x_end
86    ]
87    return result

698    pivot choice in this codebase: horizontal planes slide relative to

701    their x-coordinate (shear_y). Uses the same backward-mapping approach

703    are bilinearly interpolated, and any pixel with no corresponding
704    source coordinate within `arr`'s bounds is filled with black (fill
705    value 0).

708        arr: A 2D grayscale image array.

710        shear_y: Vertical shear factor.

723            "deformation gradient (shear_x * shear_y == 1)"

737def complex_deform(
738    arr: np.ndarray, factor_x: float = 1.0, factor_y: float = 1.0, angle: float = 0.0
739) -> np.ndarray:
740    """Apply an anisotropic stretch composed with a rotation, in one pass.
741
742    Composes a stretch (deformation gradient diag(factor_x, factor_y))
743    with a rotation (`angle` degrees, counterclockwise): the combined
744    deformation gradient is F = R(angle) @ diag(factor_x, factor_y), i.e.
745    the stretch is applied first and the rotation second. Applying both
746    in a single backward-mapping pass (rather than calling `stretch` and
747    then `rotate` separately) avoids the extra blur of interpolating
748    twice. Represents realistic loading scenarios where materials
749    experience multiple simultaneous deformation modes — typically the
750    hardest case for correlation algorithms. Pivots on the image's
751    top-left corner (0, 0), consistent with `stretch`, `translate`,
752    `rotate`, and `shear`'s pivot choice in this codebase.
753
754    Args:
755        arr: A 2D grayscale image array.
756        factor_x: Stretch factor along the x-axis, applied before the
757            rotation. Must be > 0.
758        factor_y: Stretch factor along the y-axis, applied before the
759            rotation. Must be > 0.
760        angle: Rotation angle in degrees, applied after the stretch;
761            positive is counterclockwise.
762
763    Returns:
764        A 2D uint8 array, same shape as `arr`.
765
766    Raises:
767        ValueError: If factor_x or factor_y is <= 0.
768    """
769    if factor_x <= 0:
770        raise ValueError(f"factor_x {factor_x} must be > 0")
771    if factor_y <= 0:
772        raise ValueError(f"factor_y {factor_y} must be > 0")
773
774    height, width = arr.shape
775    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
776
777    theta = np.deg2rad(angle)
778    cos_theta = np.cos(theta)
779    sin_theta = np.sin(theta)
780
781    # Backward mapping for F = R(angle) @ diag(factor_x, factor_y):
782    # F_inv = diag(1/factor_x, 1/factor_y) @ R(-angle), applied to each
783    # output coordinate to find its source. Un-rotate first, then unscale.
784    xs_rotated = cos_theta * xs + sin_theta * ys
785    ys_rotated = -sin_theta * xs + cos_theta * ys
786    xs_source = xs_rotated / factor_x
787    ys_source = ys_rotated / factor_y
788
789    return _backward_map(arr, xs_source, ys_source)

793    """Apply a discontinuous vertical-crack displacement field.
794
795    Splits the image with a vertical crack at x = width / 2: the left
796    half shifts down by `offset` pixels and the right half shifts up by
797    `offset` pixels, producing a displacement field that jumps
798    discontinuously across the crack line — unlike every other transform
799    in this module, which varies smoothly. Standard DIC assumes smooth
800    displacements and cannot capture this jump; cases like this motivate
801    the Heaviside finite-element formulation. Uses the same backward
802    mapping as the other transform functions, just with a piecewise
803    (rather than single-matrix) displacement field, so non-integer
804    source coordinates are bilinearly interpolated, and any pixel with
805    no corresponding source coordinate within `arr`'s bounds is filled
806    with black (fill value 0).
807
808    Args:
809        arr: A 2D grayscale image array.

811            shifts down by this amount, the right half shifts up by it.
812
813    Returns:
814        A 2D uint8 array, same shape as `arr`.
815    """
816    height, width = arr.shape
817    xs, ys = np.meshgrid(np.arange(width), np.arange(height))
818

820    ys_source = ys.astype(np.float64)

823    ys_source[left_half] -= offset
824    ys_source[~left_half] += offset
825
